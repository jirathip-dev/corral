## Alpha and packaging proof

Generator command used:

    python3 - <<'PY'
    import importlib.util
    from pathlib import Path
    from PIL import Image
    p=Path('tools/icon/from-user-png.py')
    s=importlib.util.spec_from_file_location('gen',p); m=importlib.util.module_from_spec(s); s.loader.exec_module(m)
    with Image.open('assets/icon/corral-icon-1024.png') as im: out=m.macos_squircle(im)
    out.save('assets/icon/corral-icon-macos.png')
    print(out.mode, out.size, [out.getpixel(p)[3] for p in ((0,0),(1023,0),(0,1023),(1023,1023),(512,512))])
    PY
